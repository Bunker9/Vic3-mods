mdp_is_expand_country = {
    OR = {
        this = c:BIC   this = c:PRU   this = c:MEX   this = c:BRZ
        this = c:PER   this = c:EGY   this = c:PAN   this = c:DEI
        this = c:SOK   this = c:SAR   this = c:SWE   this = c:DAI
        this = c:NEP   this = c:CLM   this = c:ARG   this = c:SIA
        this = c:MOR   this = c:OMA   this = c:SHW   this = c:SAF
    }
}

# -----------------------------------------------------------------------------
# Is this country in ANY power bloc (as member OR leader)?
# Evaluated on the CANDIDATE target country.
# Safe to call even if the country has no bloc (power_bloc scope simply won't exist).
# -----------------------------------------------------------------------------
mdp_is_in_power_bloc = {
    power_bloc = { exists = yes }
}

# -----------------------------------------------------------------------------
# Valid target for the GENERIC blob chain.
# Evaluated in CANDIDATE country scope; scope:mdp_attacker = the expanding country.
#
# Rule: pick the weakest recognised neighbour that:
#   - is not another expand tag
#   - is not a GP
#   - is NOT in any power bloc (member OR leader) — this is the new key guard
#     that prevents PRU/SWE/SAR from dragging a GP into the war via call_to_arms
#   - is independent (not a subject)
#   - is not already at war
#   - has at least one state to take
# -----------------------------------------------------------------------------
mdp_valid_generic_target = {
    exists = this
    is_unrecognized = no                            # recognised only
    mdp_is_expand_country = no                      # never attack another expander
    NOT = { country_rank >= rank_value:great_power } # not a GP
    is_subject = no                                 # not a formal subject
    NOT = { mdp_is_in_power_bloc = yes }            # ← KEY: not in any power bloc at all
    NOT = { this = scope:mdp_attacker }             # don't self-target
    is_at_war = no                                  # don't pile on
    any_scope_state = { always = yes }              # has states to take

     # EGY must never target Oriental Orthodox / Ethiopian-culture countries
        NOT = {
            AND = {
                scope:mdp_attacker = { this = c:EGY }
                mdp_egy_forbidden_target = yes
            }
        }
}
# -----------------------------------------------------------------------------
# EGY must never attack Oriental Orthodox / Ethiopian-culture countries.
# Evaluated in CANDIDATE country scope.
# Covers SHW, KFA (Kaffa), and any other Ethiopian-sphere minor.
# Vanilla script names:
#   religion: oriental_orthodox
#   cultures: amhara, oromo, tigrinya, somali (the Horn cultures)
# -----------------------------------------------------------------------------
mdp_egy_forbidden_target = {
    OR = {
        has_state_religion = oriental_orthodox
        country_has_primary_culture = cu:amhara
        country_has_primary_culture = cu:oromo
        country_has_primary_culture = cu:tigrinya
        country_has_primary_culture = cu:somali
    }
}

# -----------------------------------------------------------------------------
# Pre-flight check on the ATTACKER before firing mdp.1
# -----------------------------------------------------------------------------
mdp_can_blob_now = {
    mdp_is_expand_country = yes
    is_at_war = no
    infamy < 75
    any_neighbouring_country = {
        mdp_valid_generic_target = yes
    }
}
