
mdp_nepal_trigger_event  = {
	is_shown = {
		c:NEP ?= this
		NOT = { has_variable = mdp_nep_tib_done }
	}

	possible = {
		is_at_war = no
		is_active_in_diplomatic_play = no
		# China distracted: under attack, or crippled by widespread opium addiction
		c:CHI = {
			exists = yes
			OR = {
				is_at_war = yes                                # China is under attack
				has_modifier = opium_wars_addiction_modifier   # "Widespread Opium Addiction"
			}
		}
	}

	when_taken = {
		set_variable = { name = mdp_nep_tib_done value = yes }
		trigger_event = { id = mdp.1 months=6 popup = yes }
	}

	ai_chance = { value = 10000 }
}
