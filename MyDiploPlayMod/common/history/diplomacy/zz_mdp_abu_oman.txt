# HISTORY, not an event (user design rule: prefer history over day-1 events — zero engine overhead).
# Vanilla common/history/diplomacy/00_subject_relationships.txt makes ABU a GBR protectorate; this
# file (loads after, zz_ prefix) drops that pact and makes OMA the protector, so OMA can later absorb
# ABU via the mdp.3 subject-annex chain (fewer late-game nations -> smoother run).
# =============================================================================
DIPLOMACY = {
	c:OMA ?= {
		create_diplomatic_pact = {
			country = c:ABU
			type = vassal
		}
	}
}
