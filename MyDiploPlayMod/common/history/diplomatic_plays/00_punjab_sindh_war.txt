# =============================================================================
# MyDiploPlayMod — Punjab (c:PAN) starts ALREADY AT WAR with independent Sindh (c:SIN)
# at game setup. This is the correct vehicle for a start-of-game war (war = yes),
# mirroring vanilla's 00_texan_war_of_independence / 00_south_american_wars.
#   - dp_conquer_state, mutual "conquer the capital" goals.
#   - Sindh is independent at 1836 (owns STATE_SINDH, the Karachi coast) and is NOT an
#     EIC/BIC subject, so this never drags in the East India Company.
#   - Cores/infamy reduction (homeland + claim) are granted in events/diplo_test_events.txt.
# =============================================================================

DIPLOMATIC_PLAYS = {
	c:PAN ?= {
		create_diplomatic_play = {
			name = punjab_sindh_war
			target_state = s:STATE_SINDH.region_state:SIN
			war = yes
			type = dp_conquer_state

			# Punjab's goal: take Sindh's capital (the coast)
			add_war_goal = {
				holder = c:PAN
				type = conquer_state
				target_state = s:STATE_SINDH.region_state:SIN
			}
			# Sindh's counter-goal: take Punjab's capital ("each other's capital")
			add_war_goal = {
				holder = c:SIN
				type = conquer_state
				target_state = s:STATE_PUNJAB.region_state:PAN
			}
		}
	}
}
