# =============================================================================
# MyDiploPlayMod — regional power starts ALREADY AT WAR with a weak neighbour at
# game setup (war = yes — no maneuvering phase, no GP-intervention window), to kick
# off its expansion. First instance: Punjab (c:PAN) vs independent Sindh (c:SIN).
# =============================================================================

DIPLOMATIC_PLAYS = {
	c:PAN ?= {
		create_diplomatic_play = {
			name = mdp_pan_sindh_war
			target_state = s:STATE_SINDH.region_state:SIN
			war = yes
			type = dp_return_state

			# Aggressor goal: return_state = lowest-infamy full annex (base infamy 2,
			# needs the claim granted in history/states/mdp_sindh_pan_claim.txt). Sindh
			# owns one state, so this takes their whole holding.
			add_war_goal = {
				holder = c:PAN
				type = return_state
				target_state = s:STATE_SINDH.region_state:SIN
			}
			# Defender counter-goal: low-stakes humiliation (infamy 0) so it's a real
			# two-sided war without the AI overcommitting.
			add_war_goal = {
				holder = c:SIN
				type = humiliation
				target_country = c:PAN
			}
		}
		#debug_log = diplo_war_row   # commented out for master; -> test report "War & goal" table
	}
}
