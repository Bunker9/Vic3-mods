# =============================================================================
# MyDiploPlayMod — the GENERIC, event-driven blob chain (replaces the old hand-listed
# "war target order"). Run in an EXPANDER's scope from event mdp.1 (fired monthly while at
# peace). When the expander is at peace it picks its nearest WEAKEST recognised neighbour
# that isn't an expander/GP/GP-subject (mdp_valid_generic_target), gives itself the same
# incorporation package the day-1 history wars get — claim + homeland + seed pop on EVERY
# state of the target — and then declares a full-state conquest war.
#
# Dynamic war goals use vanilla's pattern (see common/scripted_effects/00_sepoy_mutiny…):
# create the play with a target_state, then re-enter it via random_diplomatic_play and add
# a conquer_state goal per remaining state.
# =============================================================================
mdp_try_next_war = {
	if = {
		limit = {
			is_at_war = no
			infamy < 75            # don't blob into a coalition death-spiral; mdp_infamy_decay helps
		}
		save_scope_as = mdp_attacker
		random_primary_culture = { save_scope_as = mdp_cul }   # attacker's culture (for homeland/pops)

		# nearest WEAKEST valid neighbour (ascending army_size → position 0 = weakest)
		ordered_neighbouring_country = {
			limit = { mdp_valid_generic_target = yes }
			order_by = army_size
			max = 1
			save_scope_as = mdp_tgt
		}

		if = {
			limit = { exists = scope:mdp_tgt }

			# 1) incorporation package on EVERY target state: claim + homeland + seed pop
			scope:mdp_tgt = {
				every_scope_state = {
					add_claim = scope:mdp_attacker
					add_homeland = scope:mdp_cul
					create_pop = { culture = scope:mdp_cul size = 10000 }
				}
			}

			# 2) declare the conquest war (capital = primary demand)
			create_diplomatic_play = {
				name = mdp_blob_war
				target_state = scope:mdp_tgt.capital
				war = yes
				type = dp_conquer_state
			}

			# 3) add a conquer_state goal for every OTHER state of the target
			random_diplomatic_play = {
				limit = {
					is_diplomatic_play_type = dp_conquer_state
					initiator = { this = scope:mdp_attacker }
					target = { this = scope:mdp_tgt }
				}
				save_scope_as = mdp_play
				scope:mdp_tgt = {
					every_scope_state = {
						limit = { NOT = { this = scope:mdp_tgt.capital } }
						save_scope_as = mdp_goalstate
						scope:mdp_play = {
							add_war_goal = {
								holder = scope:mdp_attacker
								type = conquer_state
								target_state = scope:mdp_goalstate
							}
						}
					}
				}
			}
		}
	}
}
