# =============================================================================
# MyDiploPlayMod — generic blob-chain effect.
# Scope when called: the ATTACKER country (an expand tag).
# scope:mdp_attacker must be saved by the caller (mdp.1) before this fires.
#
# NOTE on debug_log: keep strings PLAIN TEXT — square-bracket [Data.Functions]
# inside debug_log do NOT resolve here and spam ~hundreds of loc-parse errors into
# error.log (same lesson as Top40 eco_effects). Use debug_log_scopes = yes for the
# actual scope identities (Root / This / saved scopes).
# =============================================================================
mdp_try_next_war = {

    debug_log = "mdp_try_next_war START"
    debug_log_scopes = yes   # logs attacker identity + the scope stack

    # -------------------------------------------------------------------------
    # Step 1: find the WEAKEST valid neighbour (lowest population = weakest proxy).
    # order_by only works on ordered_* scopes — random_country would ignore it.
    # ordered_country + position = 0 picks the single lowest-population match.
    # -------------------------------------------------------------------------
    ordered_country = {
        limit = {
            mdp_valid_generic_target = yes   # includes the land-adjacency-to-attacker check
        }
        order_by = prestige        # ascending — lowest prestige first (valid weakest-proxy;
                                   # `country_population` is NOT a real Vic3 value — verified 0 hits.
                                   # The ~3M/large-country guard lives in the target filter via
                                   # rank_value:major_power + state-count cap, not here.)
        position = 0               # take only the weakest
        check_range_bounds = no

        save_scope_as = mdp_tgt

        debug_log = "mdp_try_next_war TARGET CHOSEN"
        debug_log_scopes = yes
    }

    # -------------------------------------------------------------------------
    # Step 2: verify we actually found a target before continuing
    # -------------------------------------------------------------------------
    if = {
        limit = { NOT = { exists = scope:mdp_tgt } }
        debug_log = "mdp_try_next_war NO VALID TARGET — aborting"
    }

    if = {
        limit = { exists = scope:mdp_tgt }

        debug_log = "mdp_try_next_war granting claims+homeland on target states"

        # ---------------------------------------------------------------------
        # Step 3: claim + homeland on every target state.
        # add_claim is a STATE-scoped effect whose VALUE is the country gaining the
        # claim — so we stay in the every_scope_state (state) scope and pass the
        # attacker as the value. (Wrapping it in scope:mdp_attacker = {} runs it in
        # COUNTRY scope and errors "Wrong scope: country, expected state_region".)
        # add_homeland is likewise a state effect taking a culture value.
        # These claims are what make the conquest LOW-INFAMY + incorporable.
        # ---------------------------------------------------------------------
        scope:mdp_tgt = {
            every_scope_state = {
                add_claim    = scope:mdp_attacker
                add_homeland = scope:mdp_attacker.ruler.culture
            }
        }

        # ---------------------------------------------------------------------
        # Step 4: seed primary-culture pops in every target state
        # ---------------------------------------------------------------------
        scope:mdp_tgt = {
            every_scope_state = {
                create_pop = {
                    culture  = scope:mdp_attacker.ruler.culture
                    religion = scope:mdp_attacker.religion
                    size     = 10000
                }
            }
        }

        # ---------------------------------------------------------------------
        # Step 5: declare the diplomatic play. We use dp_return_state (war goal
        # return_state) — base infamy 2 vs conquer_state's 5, and it reclaims a
        # CLAIMED state, which is exactly what Step 3's add_claim set up. This is
        # the lowest-infamy way to annex whole states. The play's primary war goal
        # is generated from the play-level target_state (= the capital).
        # ---------------------------------------------------------------------
        debug_log = "mdp_try_next_war declaring dp_return_state"

        scope:mdp_attacker = {
            create_diplomatic_play = {
                target_country = scope:mdp_tgt
                type           = dp_return_state
                target_state   = scope:mdp_tgt.capital   # primary goal = capital
            }
        }

        # ---------------------------------------------------------------------
        # Step 6: grab the freshly-created play and add a return_state war goal for
        # EVERY target state, including the capital — create_diplomatic_play's
        # target_state alone does not register a war goal (mirrors the hand-curated
        # zz_mdp_wars.txt pattern, which always pairs target_state with an explicit
        # add_war_goal). The capital gets primary_demand = yes; add_war_goal must run
        # in the diplomatic_play scope, so we can't loop inside create_diplomatic_play
        # — we re-enter the play here.
        # ---------------------------------------------------------------------
        random_diplomatic_play = {
            limit = {
                is_diplomatic_play_type = dp_return_state
                any_scope_play_involved = { this = scope:mdp_attacker }
                any_scope_play_involved = { this = scope:mdp_tgt }
            }
            save_scope_as = mdp_play
        }

        if = {
            limit = { exists = scope:mdp_play }
            scope:mdp_tgt = {
                every_scope_state = {
                    save_scope_as = mdp_goal_state
                    if = {
                        limit = { this = scope:mdp_tgt.capital }
                        scope:mdp_play = {
                            add_war_goal = {
                                holder       = scope:mdp_attacker
                                type         = return_state
                                target_state = scope:mdp_goal_state
                                primary_demand = yes
                            }
                        }
                    }
                    else = {
                        scope:mdp_play = {
                            add_war_goal = {
                                holder       = scope:mdp_attacker
                                type         = return_state
                                target_state = scope:mdp_goal_state
                            }
                        }
                    }
                }
            }
        }

        debug_log = "mdp_try_next_war WAR DECLARED"
        debug_log_scopes = yes
    }

    debug_log = "mdp_try_next_war END"
}
