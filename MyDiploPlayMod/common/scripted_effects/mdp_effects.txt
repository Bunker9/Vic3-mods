# =============================================================================
# MyDiploPlayMod — generic blob-chain effect.
# Scope when called: the ATTACKER country (an expand tag).
# scope:mdp_attacker must be saved by the caller (mdp.1) before this fires.
# =============================================================================
mdp_try_next_war = {

    debug_log = "[mdp_try_next_war] START — attacker: [THIS.GetName] ([THIS.GetTag])"
    debug_log_scopes = yes   # dumps full scope stack so you can verify root/this/prev

    # -------------------------------------------------------------------------
    # Step 1: find the weakest valid neighbour (lowest prestige = weakest proxy)
    # -------------------------------------------------------------------------
    random_country = {
        limit = {
            mdp_valid_generic_target = yes
            any_neighbouring_country = {
                this = scope:mdp_attacker
            }
        }
        order_by = prestige   # ascending — lowest prestige first
        max = 1

        save_scope_as = mdp_tgt

        debug_log = "[mdp_try_next_war] TARGET CHOSEN — [THIS.GetName] ([THIS.GetTag]) prestige=[THIS.GetPrestige]"
    }

    # -------------------------------------------------------------------------
    # Step 2: verify we actually found a target before continuing
    # -------------------------------------------------------------------------
    if = {
        limit = { NOT = { exists = scope:mdp_tgt } }
        debug_log = "[mdp_try_next_war] NO VALID TARGET found for [scope:mdp_attacker.GetName] — aborting"
    }

    if = {
        limit = { exists = scope:mdp_tgt }

        debug_log = "[mdp_try_next_war] Granting claims+homeland on all states of [scope:mdp_tgt.GetName]"

        # ---------------------------------------------------------------------
        # Step 3: claim + homeland on every target state
        # ---------------------------------------------------------------------
        scope:mdp_tgt = {
            every_scope_state = {
                debug_log = "[mdp_try_next_war]   -> claim+homeland on state: [THIS.GetName]"

                scope:mdp_attacker = {
                    add_claim = { target = prev }
                }
                add_homeland = { culture = scope:mdp_attacker.ruler.culture }
            }
        }

        # ---------------------------------------------------------------------
        # Step 4: seed primary-culture pops in every target state
        # ---------------------------------------------------------------------
        debug_log = "[mdp_try_next_war] Seeding pops in [scope:mdp_tgt.GetName] states (culture=[scope:mdp_attacker.ruler.culture.GetName])"

        scope:mdp_tgt = {
            every_scope_state = {
                debug_log = "[mdp_try_next_war]   -> seeding pop in state: [THIS.GetName]"
                create_pop = {
                    culture = scope:mdp_attacker.ruler.culture
                    size = 10000
                }
            }
        }

        # ---------------------------------------------------------------------
        # Step 5: declare the diplomatic play
        # ---------------------------------------------------------------------
        debug_log = "[mdp_try_next_war] Declaring diplomatic play: [scope:mdp_attacker.GetName] vs [scope:mdp_tgt.GetName]"

        scope:mdp_attacker = {
            create_diplomatic_play = {
                target_country = scope:mdp_tgt
                type = dp_conquest

                # Capital state = primary war goal
                add_war_goal = {
                    holder = this
                    type = conquer_state
                    target_state = scope:mdp_tgt.capital
                }

                # All remaining states as additional war goals
                scope:mdp_tgt = {
                    every_scope_state = {
                        limit = { NOT = { this = scope:mdp_tgt.capital } }
                        scope:mdp_attacker = {
                            add_war_goal = {
                                holder = this
                                type = conquer_state
                                target_state = prev
                            }
                        }
                    }
                }
            }
        }

        debug_log = "[mdp_try_next_war] WAR DECLARED — [scope:mdp_attacker.GetName] -> [scope:mdp_tgt.GetName] ✓"
    }

    debug_log = "[mdp_try_next_war] END — [THIS.GetName] ([THIS.GetTag])"
}
