# =============================================================================
# MyDiploPlayMod — generic blob-chain effect (PARKED 2026-06-12, pending rework).
#
# The first cut (in-game test) failed to load with three issues, so it's stubbed out:
#   1. `ordered_neighbouring_country` is NOT a real effect/list scope. Need another way to
#      find the nearest weakest neighbour — e.g. `every_country = { limit = { is_country_type
#      = recognized neighbours-me ... } }` tracking a min army_size in a var, or a different
#      list scope that actually exists.
#   2. add_claim / add_homeland inside `every_scope_state` hit a state-vs-state_region scope
#      mismatch — operate on the region_state, or use the right state scope.
#   3. create_pop got "missing culture" because scope:mdp_cul wasn't resolving — culture must
#      be captured into a saved scope that survives into the every_scope_state loop.
#
# Day-1 wars are unaffected — those live in history (common/history/diplomatic_plays/) and
# don't use this. This effect only drove the (now disabled) monthly sequential chain.
# =============================================================================
mdp_try_next_war = {
    # Find the weakest valid neighbour (lowest prestige as proxy for "weakest")
    # Save it into a named scope, then declare a full-conquest war on it.

    random_country = {
        limit = {
            mdp_valid_generic_target = yes
            any_neighbouring_country = {
                this = scope:mdp_attacker
            }
        }
        order_by = prestige   # ascending = weakest first
        max = 1

        save_scope_as = mdp_tgt

        # Grant claim + homeland on every state of target
        every_scope_state = {
            scope:mdp_attacker = {
                add_claim = { target = prev }
            }
            add_homeland = { culture = scope:mdp_attacker.ruler.culture }
        }

        # Seed pops
        every_scope_state = {
            create_pop = {
                culture = scope:mdp_attacker.ruler.culture
                size = 10000
            }
        }
    }

    # Declare the war if we found a target
    if = {
        limit = { exists = scope:mdp_tgt }
        scope:mdp_attacker = {
            create_diplomatic_play = {
                target_country = scope:mdp_tgt
                type = dp_conquest
                add_war_goal = {
                    holder = this
                    type = conquer_state
                    target_state = scope:mdp_tgt.capital
                }
                scope:mdp_tgt = {
                    every_scope_state = {
                        limit = { NOT = { this = scope:mdp_tgt.capital } }
                        scope:mdp_attacker = {
                            add_war_goal = {
                                holder = this
                                type = conquer_state
                                target_state = prev
                            }
                        }
                    }
                }
            }
        }
    }
}
