# MyDiploPlayMod - at game start, set the runtime-safe expander + subject-annexer flags ONLY.
#
# Day-1 conquest WARS live in HISTORY (gen_blob_plan.py -> zz_mdp_wars / zz_mdp_claims / zz_mdp_seed_pops):
# history start = no runtime infamy, low-infamy fully-incorporated conquest.
#
# 2026-06-30 (user): the monthly BLOB-CHAIN PULSE is REMOVED. Every expander now expands via DECISIONS
# (mdp_conquer_free_minor / mdp_annex_own_subject in mdp_expander_decisions.txt; BIC via mdp_bic_decisions),
# gated on a <4-state target + infamy < 4, with NO date conditions. This on_action now ONLY sets the startup
# flags the decisions read (mdp_is_expand_country / mdp_is_subject_annexer).
#
# on_game_started_after_lobby is ADDITIVE in Vic3 (on_actions merge) - it appends to vanilla.
# =============================================================================
on_game_started_after_lobby = { on_actions = { mdp_startup } }

# Merge-safe sub-list effect: on_actions = { ... } MERGES across mods, whereas a bare `effect = {}` on a
# shared on_action is OVERRIDDEN. Sets ONLY the runtime-safe flags the decisions gate on.
mdp_startup = {
	effect = {
		every_country = {
			limit = { mdp_is_expand_tag = yes }   # raw tag list (all live at start)
			set_variable = mdp_expander           # flag -> runtime-safe mdp_is_expand_country
		}
		# subject-annexer tags (DAI/BUR/SIA/PAN + OMA) -> mdp_is_subject_annexer. OMA starts with ABU as a
		# subject (history zz_mdp_abu_oman.txt) and annexes its own subjects via mdp_annex_own_subject.
		every_country = {
			limit = { OR = { this = c:DAI  this = c:BUR  this = c:SIA  this = c:PAN  this = c:OMA } }
			set_variable = mdp_subject_annexer
		}
		# BIC/DEI need NO startup flag - their decisions gate directly on the tag (runtime-safe).
	}
}
