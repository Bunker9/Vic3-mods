# =============================================================================
# ExpMktAccessMod — at game start, give every expand tag at least ONE trade center in
# its capital (start) state, so it has the infrastructure for global market access — the
# thing expansion is FOR. (Coastal port/shipyard seeding is a later refinement.)
# on_game_started_after_lobby is ADDITIVE in Vic3 (on_actions merge) — appends to vanilla.
# =============================================================================
on_game_started_after_lobby = {
	effect = {
		every_country = {
			limit = { expmkt_is_expand_country = yes }
			capital = {
				if = {
					limit = { NOT = { has_building = building_trade_center } }
					create_building = { building = building_trade_center reserves = 1 }
					debug_log = expmkt_trade_center_row   # commented out for master
				}
			}
		}
	}
}
