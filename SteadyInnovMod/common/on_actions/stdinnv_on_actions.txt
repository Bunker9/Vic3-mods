# =============================================================================
# SteadyInnovMod — startup tagging
# -----------------------------------------------------------------------------
# At game start, stamp every country with its starting "innovation tier":
#   - the four struggling great powers (FRA/JAP/NET/USA) get the permanent boost
#   - subjects / BIC / CHI get the heavy subject penalty
#   - other unrecognized powers get the lighter unrecognized penalty
#   - recognized non-subject middle powers get nothing (neutral baseline)
# The penalties are permanent and only cleared via the universities decision ->
# uni_event.1 (see events/uni_events.txt).
# =============================================================================
on_game_started_after_lobby = { on_actions = { stdinnv_startup } }

# Merge-safe sub-list effect: on_actions = { ... } MERGES across mods, whereas a bare
# `effect = {}` on a shared on_action is OVERRIDDEN (only the last-loaded mod's effect runs).
stdinnv_startup = {
	effect = {
		every_country = {
			if = {
				limit = {
				    OR = {
                        this = c:FRA   this = c:JAP   this = c:NET   this = c:USA
                    }
                }
				add_modifier = { name = stdinnv_boost_gp  }
			}
		}
		every_country = {
			if = {
				limit = {
				    OR = {
                        this = c:BIC
                        this = c:CHI
                        is_subject = yes
                    }
                }
				add_modifier = { name = stdinnv_subject  }
			}
			else_if = {
				limit = {
				    NOT = { is_country_type = recognized }
				}
				add_modifier = { name = stdinnv_unrecognized  }
			}
		}
	}
}
