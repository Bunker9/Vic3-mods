# =============================================================================
# SteadyInnovMod — "graduate via universities" decision
# -----------------------------------------------------------------------------
# Shown to penalised nations (and any human player). Only TAKEABLE once the
# nation has graduated out of penalty status: it must now be recognized and
# non-subject AND have 10 universities (lvl>=10) in a single state. Firing it
# triggers uni_event.1, which clears the lingering startup penalty and grants
# the catch-up reward. (A nation that is STILL unrecognized/subject sees it but
# cannot yet meet `possible` — that is intentional: graduate first.)
# =============================================================================
stdinnv_uni_built = {
	#icon = "gfx/interface/icons/timed_modifier_icons/modifier_lightbulb_positive.dds"

	is_shown = {
		OR = {
		    has_modifier = stdinnv_unrecognized
		    has_modifier = stdinnv_subject
		}
		NOT = { has_modifier = stdinnv_reward }
	}

	possible = {
		any_scope_building = {
			is_building_type = building_university
			level >= 10
		}
		NOT = { has_modifier = stdinnv_reward }
		OR = {
		    NOR = {
		        this = c:BIC
		        this = c:DEI
		        this = c:CHI
		    }
            AND = {
                OR = {
                    this = c:BIC
                    this = c:DEI
                }
                is_subject = no
            }
            AND = {
                this = c:CHI
                is_country_type = recognized
            }
		}
	}

	when_taken = {
		trigger_event = { id = uni_event.1 }
	}

	ai_chance = { value = 1000000 }
}
