# SteadyInnovMod - Treaty Port article AI-acceptance tweak
# HAND-WRITTEN. REPLACE_OR_CREATE of the vanilla treaty_port treaty article
# (common/treaty_articles/14_treaty_port.txt, Vic3 1.13.x), copied verbatim, tweaked so a
# LAPSED treaty port can be restored peacefully (no war, no infamy) when the requester still
# owns a coastal part of the state region and relations are neutral or better:
#   HOST side     - ai.inherent_accept_score +600 to give it back
#   REQUESTER side - ai.inherent_accept_score +500 to want it, proposal_weight x5 to ASK for it,
#                    evaluation_chance x3 for great powers so the article gets weighed at all.
# BRITTLE: full-object copy - re-check against vanilla after any patch touching 14_treaty_port.txt.
# The added condition mirrors this file's own input_state/state_region/any_scope_state idiom;
# relations_threshold:friendly / power_bloc / global_country_ranking are master-vetted.

REPLACE_OR_CREATE:treaty_port = {
    kind = directed
    cost = 50

    relations_progress_per_day = -0.5
    relations_improvement_min = 0

	max_target_involvement = 3000
	target_involvement_applies_to = source_country

	icon = "gfx/interface/icons/diplomatic_treaties_articles_icons/treaty_port_treaties.dds"

	flags = {
		is_treaty_port
		can_be_enforced
		ai_consider_threaten_naval_hostilities
		giftable
		hostile
	}

	usage_limit = once_per_side_with_same_inputs

	maintenance_paid_by = target_country

    required_inputs = {
        state
    }

    unlocked_by_technologies = {
        colonization
    }

	visible = {
		is_revolutionary = no
		any_scope_state = {
			is_coastal = yes
		}
	}

	possible = {
		any_scope_state = {
			is_coastal = yes
			is_potential_treaty_port = scope:other_country
		}
		scope:other_country = {
			NOT = { is_subject_of = root }
		}
	}

	requirement_to_maintain = {
		trigger = {
			exists = scope:article.input_state
		}
	}

	requirement_to_maintain = {
		trigger = {
			trigger_if = {
				limit = {
					exists = scope:is_treaty_active
				}
				scope:article.input_state ?= {
					state_region = {
						any_scope_state = {
							OR = {
								owner = scope:article.target_country
								owner = {
									OR = {
										is_secessionist = yes
										is_revolutionary = yes
										is_special_civil_war_tag = yes
									}
									is_diplomatic_play_enemy_of = scope:article.target_country
								}
							}
							controls_treaty_port_province = yes
						}
					}
				}
			}
			trigger_else = {
				always = yes
			}
		}
	}

	requirement_to_maintain = {
		trigger = {
			trigger_if = {
				limit = {
					exists = scope:is_treaty_active
				}
				scope:article.input_state ?= {
					state_region = {
						any_scope_state = {
							owner = {
								OR = {
									this = scope:article.source_country
									AND = {
										OR = {
											is_revolutionary = yes
											is_secessionist = yes
											is_special_civil_war_tag = yes
										}
										is_diplomatic_play_enemy_of = scope:article.source_country
									}
								}
							}
							is_treaty_port = no
						}
					}
				}
			}
			trigger_else = {
				always = yes
			}
		}
	}

    can_ratify = {
        scope:source_country = {
			country_rank <= scope:target_country.country_rank
        }
    }

	can_withdraw = {
		trigger_if = {
			limit = {
				scope:withdrawing_country = scope:source_country
			}
			scope:withdrawing_country = {
				country_rank > scope:non_withdrawing_country.country_rank
				OR = {
					NOT = { is_country_type = unrecognized }
					scope:target_country = { is_country_type = unrecognized }
				}
			}
		}

	}

	state_valid_trigger = {
		scope:input = {
			is_potential_treaty_port = scope:other_country
			is_treaty_port = no
		}
		NOT = {
			any_treaty = {
				any_scope_article = {
					has_type = treaty_port
					input_state = scope:input
				}
			}
		}
	}

    on_entry_into_force = {
		scope:article_options = {
			target_country = {
				save_scope_as = state_taker
			}
			source_country = {
				save_scope_as = state_giver
			}

			if = {
				limit = { exists = input_state }
				input_state = {
					hidden_effect = {
						cede_treaty_port = scope:state_taker
					}
				}
			}
		}

		scope:state_taker ?= {
			if = {
				limit = {
					has_variable = recently_traded_states_counter
				}
				change_variable = {
					name = recently_traded_states_counter
					add = 1
				}
			}
			else_if = {
				limit = {
					NOT = { has_variable = recently_traded_states_counter }
				}
				set_variable = {
					name = recently_traded_states_counter
					value = 1
				}
			}
		}


		scope:state_giver ?= {
			if = {
				limit = {
					has_variable = recently_traded_states_counter
				}
				if = {
					limit = {
						var:recently_traded_states_counter > 1
					}
					change_variable = {
						name = recently_traded_states_counter
						subtract = 1
					}
				}
				else = {
					remove_variable = recently_traded_states_counter
				}
			}
		}
	}
	on_enforced = {
		scope:article_options.input_state = {
			state_region = {
				add_claim = scope:article_options.source_country
			}
		}
	}

	on_withdrawal = {
		scope:article_options.input_state = {
			state_region = {
				add_claim = scope:article_options.source_country
			}
		}
	}
	on_break = {
		scope:withdrawing_country = {
			change_relations = {
				country = scope:non_withdrawing_country
				value = {
					add = -10
					multiply = {
						value = scope:treaty_options.remaining_binding_period
						min = 365
						divide = 365
					}
					max = -10
					min = -50
				}
			}
			change_infamy = {
				value = {
					add = 3
					multiply = {
						value = scope:treaty_options.remaining_binding_period
						min = 365
						divide = 365
					}
					max = 20
					min = 2
				}
			}
		}
	}

	ai = {
		treaty_categories = {
			expansion
		}

		# SteadyInnovMod: vanilla 0.1 means a GP almost never even weighs a treaty port.
		evaluation_chance = {
			value = 0.1

			if = {
				limit = { country_rank >= rank_value:great_power }
				multiply = 3
			}
		}

		article_ai_usage = {
            request
        }

		# SteadyInnovMod: the REQUESTER side. Push the composing AI to actually ask for a port
		# in a state region it already holds a coastal part of (its lapsed concession).
		proposal_weight = {
			value = 1

			if = {
				limit = {
					scope:article.input_state ?= {
						state_region = {
							any_scope_state = {
								owner = root
								is_coastal = yes
							}
						}
					}
				}
				multiply = 5
			}
		}

		inherent_accept_score = {
			value = 0

			if = {
				limit = {
					exists = scope:article.input_state
				}
				if = {
					limit = {
						scope:source_country ?= this
					}
					if = {
						limit = {
							country_rank < rank_value:great_power
						}
						subtract = {
							value = scope:article.input_state.ai_state_value:root
							multiply = 0.5
							desc = "DIPLOMATIC_ACCEPTANCE_TRADE_STATE_SOURCE"
						}

						if = {
							limit = {
								is_country_type = recognized
							}
							subtract = {
								value = 50
							}
						}
					}
					else = {
						subtract = {
							value = 1000
							desc = "great_power"
						}
					}

					if = {
						limit = {
							scope:article.input_state = {
								any_scope_strait = {
									exists = this
								}
							}
						}
						subtract = {
							value = 500
							desc = "concept_strait_controller"
						}
					}

					if = {
						limit = {
							country_has_isolationism_or_sakoku = yes
						}
						add = {
							desc = "law_isolationism"
							add = -1000
						}
					}

					if = {
						limit = {
							c:JAP ?= scope:source_country
							scope:source_country ?= {
								has_journal_entry = je_meiji_imperial_marriage
							}
						}
						add = {
							desc = "je_meiji_imperial_marriage"
							value = -500
						}
					}
					# SteadyInnovMod: a friendly, dominant partner reclaiming a LAPSED treaty port
					# (it already owns part of this state) faces far less resistance - offsets the
					# standard state-value unwillingness so the deal can close peacefully.
					if = {
						limit = {
							# requester already holds part of this state (a reverted treaty port)
							scope:article.input_state = {
								state_region = {
									any_scope_state = { owner = scope:target_country }
								}
							}
							scope:target_country = {
								#let Ai reclaim lapsed treaty ports
								relations:root >= relations_threshold:neutral   # neutral with us (the host)
								# THIS SO GBR can get back HK Tp from CHI
							}
						}
						add = {
							value = 600
							desc = "stdinnv_lapsed_treaty_port_desc"
						}
					}
				}
				else = {
					if = {
						limit = { "scope:article.input_state.ai_treaty_port_value(root)" > 0 }
						add = {
							value = "scope:article.input_state.ai_treaty_port_value(root)"
							desc = "DIPLOMATIC_ACCEPTANCE_TRADE_STATE_TARGET"
						}
					}
					else = {
						add = {
							value = -300
							desc = "DIPLOMATIC_ACCEPTANCE_TRADE_STATE_REJECT"
						}
					}
					if = {
						limit = {
							scope:source_country ?= {
								c:JAP ?= this
								has_journal_entry = je_iwakura_mission
								has_variable = current_expedition_location_var
								var:current_expedition_location_var ?= ROOT
							}
						}
						add = {
							desc = "HOSTING_DIPLOMATIC_MISSION"
							value = -50
						}
					}
					# SteadyInnovMod: REQUESTER side, mirror of the host's +600 below. We already hold a
					# coastal part of this region, so wanting the port back beats the standard -300 apathy.
					if = {
						limit = {
							scope:article.input_state = {
								state_region = {
									any_scope_state = {
										owner = scope:target_country
										is_coastal = yes
									}
								}
							}
							scope:source_country = {
								relations:root >= relations_threshold:neutral
							}
						}
						add = {
							value = 500
							desc = "stdinnv_reclaim_lapsed_port_desc"
						}
					}
				}
			}
		}

		wargoal_score_multiplier = {
			value = 1

			# Britain should go for Hong Kong when starting Opium Wars play against Qing
			if = {
				limit = {
					has_journal_entry = je_opium_wars
					NOT = { is_diplomatic_play_enemy_of = scope:source_country }
					scope:source_country = {
						has_variable = opium_wars_target
					}
					scope:article.input_state = {
						state_region = s:STATE_SHAOZHOU
					}
				}
				add = 100
			}
		}
	}

	wargoal = {
		execution_priority = 71
		contestion_type = control_target_state

		maneuvers = {
			value = 0

			add = {
				desc = "MANEUVERS_BASE_VALUE"
				value = 5
			}

			scope:target_country = {
				multiply = {
					desc = "MANEUVERS_TARGET_COUNTRY_POPULATION_FACTOR"
					value = total_population
					divide = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_FACTOR
					multiply = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER
                    max = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER_MAX_PER_STATE
				}
			}
		}

		infamy = {
			value = 0

			if = {
				limit = {
					exists = scope:state
				}
				scope:state = {
					save_temporary_scope_as = target_state
				}
				add = {
					desc = "INFAMY_STATE_POPULATION_FACTOR"
					value = scope:state.state_population
					divide = 10000
					max = 5
				}
				add = {
					desc = "INFAMY_STATE_GDP_FACTOR"
					value = scope:state.gdp
					divide = 50000
					max = 5
				}
			}
			min = 5
		}
	}
}
