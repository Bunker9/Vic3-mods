# ---- Decision 1: naval build up (troop ships + naval administrations + marine industry) ----
# Once-only (guard knip_navy_buildup_done). Grants the fleet + buildings via knip_naval_buildup_effect,
# then fires the marine-industry event chain (knip.1 -> knip.2).
knip_naval_buildup = {
	#icon = "gfx/interface/icons/decision_icons/decision_generic.dds"

	is_shown = {
		c:JAP ?= this
		NOT = { has_variable = knip_navy_buildup_done }
	}

	possible = {
		#TODO this is AI slop. Japan has Sakoku  law  which is a variant of law_isolationism. REDO
		NOT = { has_law = law_type:law_isolationism }
	}

	when_taken = {
		#TODO add a tooltip .. something like " dutch , spanish and portugese mothballed ships added to jap navy"
		set_variable = { name = knip_navy_buildup_done value = yes }   # load-bearing once-only guard
		knip_naval_buildup_effect = yes                                # ships + naval administrations
		trigger_event = { id = knip.1 }                                # marine-industry event chain
	}

	ai_chance = { value = 10000 }
}

# ---- Decision 2: claim Korea (gated on a real navy, i.e. decision 1 paid off) ----
# Recognized Japan with a navy larger than 30 claims every Korean state, seizes the
# Yangho port province, and drops relations with Korea to -50. Once-only.
knip_claim_korea = {
	#icon = "gfx/interface/icons/decision_icons/decision_generic.dds"

	is_shown = {
		c:JAP ?= this
		is_country_type = recognized
		NOT = { has_variable = knip_korea_claimed_done }
	}

	possible = {
		navy_size > 30                       # proxy for "decision 1 fired and gave ships"
		#NOT = { has_variable = knip_korea_claimed_done } # already done in is_shown so whats the point keep it here ? remove
	}

	when_taken = {
		save_scope_as = knip_actor
		s:STATE_SEOUL     = { add_claim = scope:knip_actor }
		s:STATE_BUSAN     = { add_claim = scope:knip_actor }
		s:STATE_YANGHO    = { add_claim = scope:knip_actor }
		s:STATE_SARIWON   = { add_claim = scope:knip_actor }
		s:STATE_PYONGYANG = { add_claim = scope:knip_actor }
		# Yangho's port tile turns Japanese; Korea keeps the rest of the state
		s:STATE_YANGHO = {
			set_owner_of_provinces = {
				country = ROOT
				provinces = { x8ACAAA }
			}
		}
		# both sides primed for war; guarded, Korea may be gone by now
		if = {
			limit = { exists = c:KOR }
			set_relations = { country = c:KOR value = -50 }
		}
		set_variable = { name = knip_korea_claimed_done value = yes }          # load-bearing guard
		# set_variable = { name = knip_fingerprint_korea_claimed value = yes }   # additive fingerprint
	}

	ai_chance = { value = 10000 }
}
