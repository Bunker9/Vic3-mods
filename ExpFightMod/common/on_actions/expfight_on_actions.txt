# =============================================================================
# ExpFightMod — at game start, arm every expand tag for fast early conquest.
# on_game_started_after_lobby is ADDITIVE in Vic3 (on_actions merge across mods),
# so this appends to vanilla's setup rather than overriding it.
#   - grant army_reserves (era_1) + general_staff (era_2): the barracks doctrine that
#     fixes low starting morale/org — also covers primitives (e.g. SOK) that lack it.
#   - apply expfight_warboost for 60 months (5 years), or 120 (10 years) for the long-war
#     expanders (SOK/OMA/SHW) that have many neighbours to conquer.
# debug_log feeds the testbook "ExpFight setup" table (commented out for master).
# =============================================================================
on_game_started_after_lobby = { on_actions = { expfight_startup } }

# Merge-safe sub-list effect: on_actions = { ... } MERGES across mods, whereas a bare
# `effect = {}` on a shared on_action is OVERRIDDEN (only the last-loaded mod's effect runs).
expfight_startup = {
	effect = {
		every_country = {
			limit = { expfight_is_expand_country = yes }
			add_technology_researched = army_reserves
			add_technology_researched = general_staff
			if = {
				limit = { expfight_is_long_war_country = yes }
				add_modifier = { name = expfight_warboost months = 120 }   # 10 yr (PAN/DEI/SOK/OMA)
			}
			else_if = {
				limit = { expfight_is_short_war_country = yes }
				add_modifier = { name = expfight_warboost months = 48 }   # 4 yr (DAI/ARG/NEP/MOR)
			}
			else = {
				add_modifier = { name = expfight_warbump months = 60 }    # 5 yr (PER/MEX/BRZ)
			}
			debug_log = expfight_armed_row   # commented out for master; -> "ExpFight setup" table
		}
	}
}
