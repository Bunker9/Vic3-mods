# COUNTRY scope. True when ROOT is a great power AND holds a treaty port anywhere in China.
# Used by the two China race decisions in common/decisions/ptm_decisions.txt.
#
# Replaces the earlier hardcoded pair (STATE_FORMOSA / STATE_SHENGJING). The race opens for
# ANY great power that has forced a concession out of the Qing, including ports taken by
# conquest or added by another mod, not just the two this mod happens to ship. NO STATE OR
# STATE-REGION IS NAMED ANYWHERE IN THIS TRIGGER - that is the point of it.
#
# "IN CHINA" = the state region still contains a region_state owned by the Qing. Testing the
# whole STATE REGION rather than the treaty-port state itself is vanilla's own idiom, and
# vanilla explains why in a comment at common/ai_strategies/00_default_strategy.txt:2019:
# "Because the target state in a treaty article is not actually the treaty port state, we need
# to check the entire state region". The treaty port is a separate region_state carved out of
# the host's, so the host is a SIBLING region_state, not the port's owner.
#
# This is preferred over a strategic-region sweep (sr:region_north_china / sr:region_south_china):
# those two miss Shengjing, which sits in sr:region_northeast_asia, and patching that hole meant
# either naming Shengjing outright - the hardcode this trigger exists to remove - or reaching for
# geographic_region_manchuria_old, which also drags in the four Korean states.
#
# country_rank >= rank_value:great_power is how vanilla's own country_is_great_power
# scripted trigger is written (common/scripted_triggers/00_ip4_victoria_scripted_triggers.txt:297).
# THE custom_tooltip WRAPPER IS NOT OPTIONAL. Without it the engine auto-generates the
# requirement text in the decision panel by walking the raw trigger tree, and a doubly nested
# list scope (any_scope_state -> state_region -> any_scope_state) renders as an unreadable blob
# of "any state where any state in its state region where owner is China". Wrapping the whole
# condition in one custom_tooltip collapses it to a single readable line and the trigger still
# evaluates normally - the conditions inside AND together exactly as they would bare.
# Vanilla does this for the same reason on every non-trivial scripted trigger, including the
# country_is_great_power referenced below and is_in_civil_war
# (common/scripted_triggers/00_scripted_triggers.txt:1015), which wraps a list scope just like
# this one does.
ptm_is_great_power_with_a_china_treaty_port = {
	custom_tooltip = {
		text = ptm_gp_china_treaty_port_tt
		country_rank >= rank_value:great_power
		any_scope_state = {
			is_treaty_port = yes
			state_region = {
				any_scope_state = {
					owner = c:CHI
				}
			}
		}
	}
}

# -----------------------------------------------------------------------------
# LAPSED-PORT RECLAIM.
# ptm_port_is_lapsed_for_root: $PTM_STATE$ = a concession state region key. True when ROOT (the
# original grantee) no longer holds a treaty port in that region AND another ELIGIBLE country now
# owns a coastal, port-eligible part of it. Matches take_treaty_port's valid{} (owner = target_country,
# is_potential_treaty_port = root; war_goal_types/25_take_treaty_port.txt).
# NO is_country_type check on the owner: a lapsed port is always held by a CENTRALIZED nation
# (host, conqueror or released tag) - decentralized nations neither conquer nor hold ports, they
# exist only to be colonized (hk vic3-token-gotchas.md); dp_take_treaty_port.possible blocks them too.
# OWNER MUST BE ELIGIBLE: not our own subject, and not already sharing our market / customs union
# (a treaty port only grants market access - it adds nothing against a market partner, and the
# war goal would be pointless). These exclusions nullify both TP eligibility and this decision.
# RELATIONS BELOW NEUTRAL: at neutral or better the treaty route should win, so the war route
# hides itself (SteadyInnovMod's treaty_port article boosts asking + giving from neutral up).
ptm_port_is_lapsed_for_root = {
	NOT = { s:$PTM_STATE$ = { any_scope_state = { owner = root  is_treaty_port = yes } } }
	s:$PTM_STATE$ = {
		any_scope_state = {
			is_coastal = yes
			is_potential_treaty_port = root
			owner = {
				NOT = { this = root }
				NOT = { is_subject_of = root }
				NOT = { market = root.market }              # same market area (covers customs unions, subject markets)
				NOT = { is_in_customs_union_with = root }   # explicit customs-union partner
				relations:root < relations_threshold:neutral # war route only when relations are sour
			}
		}
	}
}

# Does ROOT have ANY lapsed concession port? Enumerated for all eight PTM colonizers, mirroring
# ptm_settle_concession_pops' per-tag branching (no runtime fingerprint). Pairs match ptm_treaties.txt.
ptm_has_any_lapsed_treaty_port = {
	OR = {
		AND = { c:FRA ?= this  OR = { ptm_port_is_lapsed_for_root = { PTM_STATE = STATE_FORMOSA }  ptm_port_is_lapsed_for_root = { PTM_STATE = STATE_TONKIN } } }
		AND = { c:RUS ?= this  ptm_port_is_lapsed_for_root = { PTM_STATE = STATE_SHENGJING } }
		AND = { c:NET ?= this  ptm_port_is_lapsed_for_root = { PTM_STATE = STATE_FARS } }
		AND = { c:BEL ?= this  OR = { ptm_port_is_lapsed_for_root = { PTM_STATE = STATE_OMAN }  ptm_port_is_lapsed_for_root = { PTM_STATE = STATE_BAS_CONGO } } }
		AND = { c:SWE ?= this  ptm_port_is_lapsed_for_root = { PTM_STATE = STATE_SOUTH_MADAGASCAR } }
		AND = { c:PRU ?= this  ptm_port_is_lapsed_for_root = { PTM_STATE = STATE_ANTIOQUIA } }
		AND = { c:AUS ?= this  ptm_port_is_lapsed_for_root = { PTM_STATE = STATE_NORTH_BORNEO } }
		AND = { c:USA ?= this  OR = { ptm_port_is_lapsed_for_root = { PTM_STATE = STATE_HAIL }  ptm_port_is_lapsed_for_root = { PTM_STATE = STATE_DONGOLA } } }
	}
}
