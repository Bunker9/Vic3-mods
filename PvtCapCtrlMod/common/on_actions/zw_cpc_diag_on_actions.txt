# PvtCapCtrlMod — DIAGNOSTIC PROBE. DELETE (or comment debug_log) BEFORE MASTER.
# This file uses ONLY standard Vic3 (on_actions + scripted effect + debug_log) — NO INJECT_OR_CREATE.
# So it runs even if the INJECT/REPLACE directives fail to parse, which lets it split the two
# possible failure modes if industry still over-stacks after a run (needs -debug_mode):
#
#   * If debug.log shows "cap var FOUND on region" + "value ... CORRECT"  -> the history set_variable
#     AND the gates' read path (state_region.var:zw_mfg_cap) BOTH work. So persistent stacking means
#     the INJECT_OR_CREATE / REPLACE hooks themselves did not apply  ==> FAILURE MODE #1 (directives).
#   * If it shows "cap var MISSING on region" (and/or "on INSTANCE")    -> the var did not land where
#     the gates read it                                                  ==> FAILURE MODE #2 (var scope).
#   * If it shows NOTHING (not even "diag start")                        -> mod not loading, or debug off.
on_game_started_after_lobby = { on_actions = { zw_cpc_diag } }

# Merge-safe sub-list effect (on_actions = {} merges across mods; a bare effect = {} would be overridden).
zw_cpc_diag = {
	effect = {
		debug_log = "pcc - diag - start (reading caps the same way the gates do)"

		# Midlands (GBR) should be a pop-scaled cap of 107 (2.58M / 24000).
		s:STATE_MIDLANDS.region_state:GBR ?= {
			if = {
				limit = { state_region = { has_variable = zw_mfg_cap } }
				debug_log = "pcc - diag - MIDLANDS cap var FOUND on region"
				if = {
					limit = { state_region = { var:zw_mfg_cap = 107 } }
					debug_log = "pcc - diag - MIDLANDS cap value = 107 CORRECT"
					# FINGERPRINT (variable; DEV-RULES "Save-fingerprint debugging") - additive, no logic change.
					# Durable save-side proof the Midlands cap landed correct (=107), readable offline with
					# no -debug_mode log run. Comment out at Ceremony 3 promote.
					set_variable = { name = zw_fingerprint_midlands_cap_107 value = yes }
				}
			}
			else = { debug_log = "pcc - diag - MIDLANDS cap var MISSING on region" }
			if = {
				limit = { has_variable = zw_mfg_cap }
				debug_log = "pcc - diag - MIDLANDS cap var on INSTANCE not region"
			}
			if = {
				limit = { zw_sv_state_mfg_levels >= 0 }
				debug_log = "pcc - diag - MIDLANDS mfg-sum script value reads OK"
			}
		}

		# Jiangxi (CHI) should be the highest cap, 1008.
		s:STATE_JIANGXI.region_state:CHI ?= {
			if = {
				limit = { state_region = { var:zw_mfg_cap = 1008 } }
				debug_log = "pcc - diag - JIANGXI cap value = 1008 CORRECT"
				# FINGERPRINT (variable; DEV-RULES "Save-fingerprint debugging") - additive, no logic change.
				# Durable save-side proof Jiangxi got the highest cap (=1008). Comment out at promote.
				set_variable = { name = zw_fingerprint_jiangxi_cap_1008 value = yes }
			}
			else = { debug_log = "pcc - diag - JIANGXI cap value WRONG or unreadable on region" }
		}
	}
}
