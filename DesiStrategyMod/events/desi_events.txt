# =============================================================================
# DesiStrategyMod - hidden effect-engine events (2026-07-10).
# desi_event.1 = the "Integrate the Frontier" (desi_incorporate_state) payload.
# WHY an event: a decision's when_taken is re-computed during tooltip preview,
# and the var-fed `multiplier = var:desi_incorporation_burden` evaluated to
# 'none' on every hover - the long-standing "Value of wrong type" spam noted in
# the decision's TODO. A hidden triggered event's immediate never runs in
# preview (same fix as SteadyInnov uni_event.4, in-game verified 2026-07-10).
# =============================================================================
desi_event.1 = {
	type = country_event
	hidden = yes

	immediate = {
		# guard-init the cumulative burden (LOAD-BEARING, drives the multiplier) once
		# (set-before-change; the has_variable guard is also the dummy-use reader).
		if = {
			limit = { NOT = { has_variable = desi_incorporation_burden } }
			set_variable = { name = desi_incorporation_burden value = 0 }
		}
		#if = { #TODO fingerprint toggle script missed taking care of empty if stmt
			# limit = { NOT = { has_variable = desi_fingerprint_incorporated } }
			# set_variable = { name = desi_fingerprint_incorporated value = 0 }
		#} #TODO fingerprint toggle script missed taking care of empty if stmt closing braces

		# Incorporate EVERY eligible unincorporated state at once (the just-conquered
		# Indian lands). For each, add its size (millions) to the running burden
		# (scoped back to ROOT = the country).
		every_scope_state = {
			limit = {
				is_incorporated = no
				OR = {
					region = sr:region_north_india
					region = sr:region_south_india
					region = sr:region_himalayas
					region = sr:region_indochina
					region = sr:region_greater_persia
				}
			}
			set_state_type = incorporated
			save_scope_as = desi_inc_state
			ROOT = {
				change_variable = { name = desi_incorporation_burden add = desi_incorporation_state_millions }
				# change_variable = { name = desi_fingerprint_incorporated add = 1 }
			}
		}
		# RECALC the single cost modifier scaled to the cumulative burden
		# (remove + re-add => never spams modifier stacks).
		if = {
			limit = { has_modifier = modifier_desi_incorporation_cost }
			remove_modifier = modifier_desi_incorporation_cost
		}
		add_modifier = {
			name = modifier_desi_incorporation_cost
			months = 36
			multiplier = var:desi_incorporation_burden
		}
	}
}

# -----------------------------------------------------------------------------
# desi_event.3 - Gandhi spawn payload for desi_gandhi_in_africa.
# gandhi_spawn is VANILLA's own guard (india_nationalism_events.txt:470,476), so
# setting it here retires vanilla's SAF-only event .4 for good.
# Vanilla template is deliberate: .5, the pan-national movement and the
# achievement all match on has_template, so the 1910 move to BIC still works.
# Popularity: template gives charismatic 50 + inspirational_orator 25, and
# masterful_diplomat adds 50, so he sits at the MAX_POPULARITY cap of 100.
# -----------------------------------------------------------------------------
desi_event.3 = {
	type = country_event
	hidden = yes

	immediate = {
		set_global_variable = gandhi_spawn

		# The African state that qualified us; carries the congress modifier.
		random_scope_state = {
			limit = {
				is_in_geographic_region = geographic_region_africa
				any_scope_pop = {
					religion = rel:hindu
					culture = { has_discrimination_trait_group = heritage_group_south_asian }
				}
			}
			save_scope_as = desi_gandhi_state
			add_modifier = {
				name = modifier_natal_indian_congress
				days = long_modifier_time
			}
		}

		create_character = {
			template = BIC_mohandas_karamchand_gandhi
			save_scope_as = desi_gandhi_scope
		}
		scope:desi_gandhi_scope = {
			add_trait = masterful_diplomat
		}

		# debug_log = "desi - gandhi-africa - Gandhi spawned in an African state"

		# Boolean fp: the decision is once-only, so a counter could only read 1.
		# if = {
			# limit = { NOT = { has_variable = desi_fingerprint_gandhi_spawn } }
			# set_variable = { name = desi_fingerprint_gandhi_spawn value = yes } # BDD(DESI-GANDHI) for gandhi-africa
		# }
	}
}

# -----------------------------------------------------------------------------
# desi_event.4 - Gandhi leads India payload for desi_gandhi_leads_india.
# Three-way search copied from vanilla india_nationalism_events.5:570-611: take
# him off whichever country holds him, else out of the exile pool, else create
# him. gandhi_spawn is set here too, so calling him home also retires vanilla .4.
# -----------------------------------------------------------------------------
desi_event.4 = {
	type = country_event
	hidden = yes

	immediate = {
		set_variable = { name = desi_gandhi_leads_done value = yes }
		set_global_variable = gandhi_spawn

		if = {
			limit = {
				any_country = {
					any_scope_character = { has_template = BIC_mohandas_karamchand_gandhi }
				}
			}
			random_country = {
				limit = {
					any_scope_character = { has_template = BIC_mohandas_karamchand_gandhi }
				}
				random_scope_character = {
					limit = { has_template = BIC_mohandas_karamchand_gandhi }
					transfer_character = ROOT
				}
			}
		}
		else_if = {
			limit = {
				any_character_in_exile_pool = { has_template = BIC_mohandas_karamchand_gandhi }
			}
			random_character_in_exile_pool = {
				limit = { has_template = BIC_mohandas_karamchand_gandhi }
				transfer_character = ROOT
				hidden_effect = { free_from_exile_effect = yes }
			}
		}
		else = {
			create_character = { template = BIC_mohandas_karamchand_gandhi }
		}

		# He is in ROOT now whichever branch ran, so find him once and promote.
		random_scope_character = {
			limit = { has_template = BIC_mohandas_karamchand_gandhi }
			save_scope_as = desi_gandhi_leader
		}
		scope:desi_gandhi_leader = {
			if = {
				limit = { NOT = { has_trait = masterful_diplomat } }
				add_trait = masterful_diplomat
			}
			set_as_interest_group_leader = yes
		}

		# set_character_as_ruler carries a documented CAUTION against chartered
		# companies: "the character's country will be affected" (effects.log:12754).
		# BIC is one (00_subject_relationships.txt:134), so it is guarded, and the
		# Company gets him as IG leader only. T192 covers testing the unguarded case.
		if = {
			limit = { NOT = { is_subject_type = subject_type_chartered_company } }
			scope:desi_gandhi_leader = {
				set_character_as_ruler = yes
			}
		}

		# debug_log = "desi - gandhi-leads - Gandhi installed as IG leader"

		# Boolean fp: once-only decision, so a counter could only read 1.
		# if = {
			# limit = { NOT = { has_variable = desi_fingerprint_gandhi_leads } }
			# set_variable = { name = desi_fingerprint_gandhi_leads value = yes } # BDD(DESI-GANDHI) for gandhi-leads
		# }
	}
}
