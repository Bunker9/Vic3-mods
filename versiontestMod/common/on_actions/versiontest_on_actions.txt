# =============================================================================
# versiontestMod — fire the load-confirmation popup on game start, to everyone.
# Additive on_actions sub-list pattern (does NOT add a 2nd `effect` to vanilla).
# NOTE: on_actions do NOT hotload — changes need a return to the main menu.
# =============================================================================

on_game_started_after_lobby = {
	on_actions = { versiontest_startup }
}

versiontest_startup = {
	effect = {
		every_country = {
			trigger_event = { id = versiontest.1 }
		}
	}
}
