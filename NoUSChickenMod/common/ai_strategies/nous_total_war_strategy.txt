# NoUSChickenMod - "never white-peace" diplomatic AI strategy for the USA. HAND-WRITTEN.
# Eligible ONLY for the USA while the War Bonds window is active (flag nous_warbonds_on, set by
# nous_warbonds_effects.txt), with a high weight so the USA actually adopts it during that window.
# It adds to diplomatic_play_boldness - the token vanilla uses for its "never surrender" cases
# (ai_strategy_default: Taiping vs China, Caucasus states vs Russia, etc. all add = 1000 keyed on
# is_diplomatic_play_enemy_of). Strategy token scores are ADDITIVE across active strategies (per the
# vanilla base-strategy comment), so this stacks on top of the default and makes the USA refuse to back
# down vs Mexico or in the civil war, instead of seeking a white peace.
ai_strategy_nous_total_war = {
	icon = "gfx/interface/icons/ai_strategy_icons/armed_isolationism.dds"

	type = diplomatic

	possible = {
		c:USA ?= this
		has_country_flag nous_warbonds_on
	}

	weight = {
		value = 1000
	}

	# Do not back down in diplomatic plays vs Mexico or while in civil war (mirrors vanilla precedent).
	diplomatic_play_boldness = {
		value = 1
		if = {
			limit = {
				OR = {
					is_diplomatic_play_enemy_of = c:MEX
					is_in_civil_war = yes
				}
			}
			add = 1000
		}
	}

	# Believe it can win, so it presses its wargoals to conclusion rather than seeking peace. TUNABLE.
	recklessness = {
		value = 1
		if = {
			limit = {
				OR = {
					is_diplomatic_play_enemy_of = c:MEX
					is_in_civil_war = yes
				}
			}
			add = 100
		}
	}
}
