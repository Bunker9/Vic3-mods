# Top40EcoBoostMod - oil & rubber partnership payloads. HAND-WRITTEN.
#
# REWORK 2026-08-09: driven by the two decisions in common/decisions/eco_oilrubber_decisions.txt,
# not by on_acquired_technology. ROOT is the DISCOVERER (the country that took the decision) in
# every event here, which is what lets add_ownership hand the new RGOs to it - the same
# root-owns-what-it-funds shape as PrettyTreatyMod's ptm_event.3 China mines.
#
# KEPT FROM THE 2026-06-16 FIX: oil_rig / rubber_plantation are DISCOVERABLE RGOs, so the
# partner's deposits must be revealed with force_resource_discovery BEFORE anything can be
# built, and the build must wait a tick - in the same tick as the tech grant can_construct
# still reads false. Hence the 5-day deferral to ecoor.3/.4.
# =============================================================================

# ---- OIL deal (Root = the discoverer). Oman is the partner. ----
ecoor.1 = {
	type = country_event
	hidden = yes

	immediate = {
		#debug_log = "eco - oil-partnership - deal struck"
		set_global_variable = { name = eco_oil_deal_done value = yes }

		c:OMA ?= {
			if = {
				limit = { NOT = { has_technology_researched = pumpjacks } }
				add_technology_researched = pumpjacks
			}
			every_scope_state = {
				limit = { has_potential_resource = building_oil_rig }
				force_resource_discovery = building_oil_rig
			}
		}

		# The bargain: Oman ships the oil and grants trade privileges, the discoverer pays,
		# guarantees military assistance, and carries an obligation for the favour.
		create_treaty = {
			name = treaty_name_eco_oil_concord
			first_country = root
			second_country = c:OMA
			is_draft = no
			binding_period = { years = 5 }
			articles_to_create = {
				{
					article = trade_privilege
					source_country = c:OMA
					target_country = root
				}
				{
					article = goods_transfer
					source_country = c:OMA
					target_country = root
					inputs = {
						{ goods = g:oil }
						{ quantity = 100 }
					}
				}
				{
					article = money_transfer
					source_country = root
					target_country = c:OMA
					inputs = {
						{ quantity = 4000 }
					}
				}
				{
					article = military_assistance
					source_country = root
					target_country = c:OMA
				}
			}
		}
		set_owes_obligation_to = { country = c:OMA setting = yes }

		trigger_event = { id = ecoor.3 days = 5 }    # build after the discovery registers
		trigger_event = { id = ecoor.5 days = 90 }   # combustion engine follow-up
	}
}

# ---- OIL build, 5 days on (Root = the discoverer, who ends up owning the rigs) ----
ecoor.3 = {
	type = country_event
	hidden = yes

	immediate = {
		#debug_log = "eco - oil-partnership - rigs built"
		c:OMA ?= {
			every_scope_state = {
				limit = { can_construct_building = building_oil_rig }
				create_building = {
					building = building_oil_rig
					add_ownership = { country = { country = root levels = 2 } }
					reserves = 1
				}
			}
		}
	}
}
