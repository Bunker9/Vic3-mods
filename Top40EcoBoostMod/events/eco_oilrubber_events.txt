namespace = ecoor

# =============================================================================
# Top40EcoBoostMod - oil & rubber reward chain (REDESIGN 2026-06-15).
#
# WRONG (old): a day-1 history file create_building'd oil_rig / rubber_plantation. Those
# are DISCOVERABLE RGOs gated behind techs (oil_rig <- pumpjacks, rubber_plantation <-
# rubber_mastication) and are NOT constructible at the 1836 start, so nothing built.
#
# RIGHT (now): common/on_actions/eco_oilrubber_on_actions.txt hooks on_acquired_technology.
# The FIRST time ANY country in the world researches the unlocking tech, we gift that tech to
# the themed beneficiaries (oil -> OMA ; rubber -> BRZ/SIA, SIA = Thailand/Siam) and seed up to 5 of the RGO
# (1 per highest-gdp constructible state) via eco_seed_spread_n. Fired ON the beneficiary so
# ROOT = beneficiary (eco_seed_spread_n relies on ROOT).
# =============================================================================

# ---- OIL grant (Root = beneficiary, e.g. OMA) ----
ecoor.1 = {
	type = country_event
	hidden = yes

	immediate = {
		debug_log = "ecoor.1 oil grant FIRED"
		debug_log_scopes = yes
		if = {
			limit = { NOT = { has_technology_researched = pumpjacks } }
			add_technology_researched = pumpjacks
		}
		eco_seed_spread_n = { B = building_oil_rig N = 5 }   # max 5, 1 per best constructible state
	}
}

# ---- RUBBER grant (Root = beneficiary, e.g. BRZ/THI/SIA) ----
ecoor.2 = {
	type = country_event
	hidden = yes

	immediate = {
		debug_log = "ecoor.2 rubber grant FIRED"
		debug_log_scopes = yes
		if = {
			limit = { NOT = { has_technology_researched = rubber_mastication } }
			add_technology_researched = rubber_mastication
		}
		eco_seed_spread_n = { B = building_rubber_plantation N = 5 }
	}
}
