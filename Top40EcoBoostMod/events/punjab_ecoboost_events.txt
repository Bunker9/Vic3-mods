namespace = punjab_ecoboost

# =============================================================================
# Top40EcoBoostMod — Mod 1 economic bootstrap: PUNJAB (c:PAN)
# Champ = opium (throughput +200% + SoL), seeded toward 2X=10 over 5 years.
# Support = plantations (silk/cotton seeded toward X=5; dye/tobacco already > cap).
# Flavour = 1 each, day 1 (adds the missing artillery + paper; iron/wood/tools/
#           small-arms already present at start). No flavour modifier yet.
# Tech granted FIRST (gunsmithing → artillery foundry building; artillery → cannon
# artillery unit) so the foundry can be built and the army can consume artillery.
# All placement is dynamic (ordered_scope_state) — no hardcoded state IDs.
# v1 NOTE: yearly adds are a fixed +1/yr for 5 yrs (no per-good country cap guard
#          yet) — will overshoot the 10/5 caps slightly; cap-checking is a TODO.
# =============================================================================

# ---- Day-1 setup -----------------------------------------------------------
punjab_ecoboost.1 = {
	type = country_event
	hidden = yes

	immediate = {
		# 1) TECH FIRST
		add_technology_researched = gunsmithing
		add_technology_researched = artillery

		# 2) CHAMP throughput modifier (opium), Phase 1 = 20 years
		add_modifier = { name = mod1_champ_boost months = 240 }

		# 3) Phase-1 SoL (+5) to all incorporated states, 20 years
		every_scope_state = {
			limit = { is_incorporated = yes }
			add_modifier = { name = mod1_state_sol_p1 months = 240 }
		}

		# 4) FLAVOUR day-1: add the two missing buildings (artillery foundry, paper mill)
		ordered_scope_state = {
			limit = { is_incorporated = yes NOT = { has_building = building_artillery_foundry } }
			order_by = gdp
			position = 0
			check_range_bounds = no
			create_building = {
				building = "building_artillery_foundry"
				add_ownership = { country = { country = ROOT levels = 1 } }
				reserves = 1
				activate_production_methods = { "pm_cannons" }
			}
		}
		ordered_scope_state = {
			limit = { is_incorporated = yes NOT = { has_building = building_paper_mill } }
			order_by = gdp
			position = 0
			check_range_bounds = no
			create_building = {
				building = "building_paper_mill"
				add_ownership = { country = { country = ROOT levels = 1 } }
				reserves = 1
			}
		}

		# 5) start the 5-year champ/support seeding pulse
		set_variable = { name = eco_year value = 0 }
		trigger_event = { id = punjab_ecoboost.2 years = 1 }

		# 6) schedule the Phase-2 SoL swap at year 20
		trigger_event = { id = punjab_ecoboost.3 years = 20 }
	}
}

# ---- Yearly champ/support seeding (fires ~5 times) -------------------------
punjab_ecoboost.2 = {
	type = country_event
	hidden = yes

	immediate = {
		# CHAMP: +1 opium / year (toward 2X = 10)
		ordered_scope_state = {
			limit = { is_incorporated = yes has_building = building_opium_plantation }
			order_by = gdp
			position = 0
			check_range_bounds = no
			add_building_level = { building = building_opium_plantation level = 1 }
		}
		# SUPPORT: +1 / year on the under-cap supports (silk, cotton)
		ordered_scope_state = {
			limit = { is_incorporated = yes has_building = building_silk_plantation }
			order_by = gdp
			position = 0
			check_range_bounds = no
			add_building_level = { building = building_silk_plantation level = 1 }
		}
		ordered_scope_state = {
			limit = { is_incorporated = yes has_building = building_cotton_plantation }
			order_by = gdp
			position = 0
			check_range_bounds = no
			add_building_level = { building = building_cotton_plantation level = 1 }
		}

		# advance year; reschedule until 5 pulses done
		change_variable = { name = eco_year add = 1 }
		if = {
			limit = { var:eco_year < 5 }
			trigger_event = { id = punjab_ecoboost.2 years = 1 }
		}
	}
}

# ---- Phase-2 SoL (permanent) at year 20 ------------------------------------
punjab_ecoboost.3 = {
	type = country_event
	hidden = yes

	immediate = {
		every_scope_state = {
			limit = { is_incorporated = yes }
			add_modifier = { name = mod1_state_sol_p2 }   # no duration = permanent
		}
	}
}
