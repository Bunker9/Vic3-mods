# 1. OIL. Whoever first researches pumpjacks partners with Oman.
eco_oil_partnership = {

	is_shown = {
		has_technology_researched = pumpjacks
		NOT = { has_global_variable = eco_oil_deal_done }
		exists = c:OMA
	}

	possible = {
		is_at_war = no
		custom_tooltip = {
			text = eco_oil_partnership_gate_tt
			c:OMA = { is_at_war = no }
		}
	}

	when_taken = {
		custom_tooltip = { text = eco_oil_partnership_tt }
		trigger_event = { id = ecoor.1 }
	}

	ai_chance = { value = 10000 }
}
