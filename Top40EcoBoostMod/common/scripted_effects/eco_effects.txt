# =============================================================================
# Top40EcoBoostMod — generic eco-boost effects + per-nation config.
#
# PLACEMENT RULE (eco_seed_capped) — v2, user 2026-06-10:
#   Bring a building up to its cap N, building only the DEFICIT (need = N - already-owned),
#   so a nation that already owns some of the good is NOT overshot. The deficit is spread:
#     1) one level in each constructible state, MOST-PRODUCTIVE (highest gdp) first,
#     2) any remainder (need > #states) added to the single most-productive state.
#   Each state's new level = its CURRENT level + its share, so create_building (level =
#   max(scripted, existing)) ADDS exactly `share`. Final total = exactly the cap N.
#   Caps (roadmap): champ = 2*X, support = 1*X each.  X: PAN=5, PRU=7.
#   FLAVOUR is DIFFERENT (eco_seed_flavour): ALWAYS +1 in the best state, additive,
#   regardless of existing — a guaranteed seed, not a cap.
#
# Logging: debug_log = $B$ (building name header) + debug_log_scopes (state identity)
#   land in debug.log. (Data-function loc in debug_log does NOT resolve — removed.)
# State "flags": eco_<type>_here + eco_best_here (state vars), cleared by
#   eco_cleanup_flags. eco_placed is a COUNTRY var, reset at the top of each call.
# =============================================================================

# ---- grant the military-readiness techs (line infantry + artillery) ---------
eco_grant_military_techs = {
	add_technology_researched = line_infantry   # line infantry units
	add_technology_researched = artillery       # cannon artillery units (pm_cannons)
	add_technology_researched = gunsmithing     # unlocks the artillery foundry building
}

# ---- CAPPED SPREAD: place N levels of $B$, best-first, remainder to best state -
# params: $B$ = building type, $TYPE$ = champ|support|flavour, $N$ = target levels
eco_seed_capped = {
	#debug_log = $B$                                  # header: building display name
	#debug_log_scopes = yes                           # logs the country (Root) even if need=0

	# 0) how many levels of $B$ does the country ALREADY own? (needed = cap - have)
	#    Reported via one ECO_HAVE marker per pre-existing level.
	set_variable = { name = eco_have value = 0 }
	every_scope_state = {
		limit = { has_building = $B$ }
		save_temporary_scope_value_as = { name = eco_lvl value = b:$B$.level }
		ROOT = { change_variable = { name = eco_have add = scope:eco_lvl } }
	}
	set_variable = { name = eco_mark value = 0 }
	while = {
		limit = { var:eco_mark < var:eco_have }
		#debug_log = eco_have_$TYPE$
		change_variable = { name = eco_mark add = 1 }
	}

	# 1) NEED = cap - have. Build ONLY the deficit so the total reaches exactly the cap
	#    (don't overshoot when the nation already owns some of the good).
	set_variable    = { name = eco_need value = $N$ }
	change_variable = { name = eco_need subtract = var:eco_have }
	if = { limit = { var:eco_need < 0 } set_variable = { name = eco_need value = 0 } }

	# 2) count constructible states (S); best-state share = 1 + remainder when NEED > S.
	set_variable = { name = eco_S value = 0 }
	every_scope_state = {
		limit = { can_construct_building = $B$ }
		ROOT = { change_variable = { name = eco_S add = 1 } }
	}
	set_variable = { name = eco_bestlevel value = 1 }
	if = {
		limit = { var:eco_S > 0  var:eco_S < var:eco_need }
		set_variable    = { name = eco_bestlevel value = var:eco_need }
		change_variable = { name = eco_bestlevel subtract = var:eco_S }
		change_variable = { name = eco_bestlevel add = 1 }
	}

	# 3) place the deficit, highest-gdp first. Each state's target level = its CURRENT level
	#    + its share (best state = the bulk, others = 1), so create_building ADDS `share`
	#    levels (level = max(scripted, existing)). Capped at NEED total new levels.
	set_variable = { name = eco_placed value = 0 }
	ordered_scope_state = {
		order_by = gdp
		limit = { can_construct_building = $B$ }
		if = {
			limit = { ROOT = { var:eco_placed < var:eco_need } }
			set_variable = eco_$TYPE$_here
			# this state's share (country var so the marker loop can read it)
			if = {
				limit = { ROOT = { var:eco_placed = 0 } }        # BEST (first) state — the bulk
				set_variable = eco_best_here
				ROOT = { set_variable = { name = eco_share value = var:eco_bestlevel } }
			}
			else = {                                             # other states — 1 level
				ROOT = { set_variable = { name = eco_share value = 1 } }
			}
			# target level = current level here (0 if none) + share -> adds exactly `share`
			set_variable = { name = eco_cur value = 0 }
			if = { limit = { has_building = $B$ } set_variable = { name = eco_cur value = b:$B$.level } }
			change_variable = { name = eco_cur add = ROOT.var:eco_share }
			create_building = { building = $B$ level = var:eco_cur reserves = 1 }
			ROOT = { change_variable = { name = eco_placed add = var:eco_share } }
			# census: one ECO_PLACED marker per LEVEL actually added in this state
			ROOT = {
				set_variable = { name = eco_mark value = 0 }
				while = {
					limit = { var:eco_mark < var:eco_share }
					#debug_log = eco_hit_$TYPE$
					change_variable = { name = eco_mark add = 1 }
				}
			}
			#debug_log_scopes = yes                               # logs this state's identity
		}
	}
}

# ---- FLAVOUR: ALWAYS add 1 level of $B$ in the single most-productive (highest-gdp)
#      constructible state, REGARDLESS of what the nation already owns. A guaranteed seed
#      (so AI construction chains don't fail), NOT a cap — it does not subtract existing.
# params: $B$ = building type, $TYPE$ = flavour
eco_seed_flavour = {
	#debug_log = $B$
	#debug_log_scopes = yes                           # logs the country (Root)

	# report pre-existing levels (ECO_HAVE) for the census
	set_variable = { name = eco_have value = 0 }
	every_scope_state = {
		limit = { has_building = $B$ }
		save_temporary_scope_value_as = { name = eco_lvl value = b:$B$.level }
		ROOT = { change_variable = { name = eco_have add = scope:eco_lvl } }
	}
	set_variable = { name = eco_mark value = 0 }
	while = {
		limit = { var:eco_mark < var:eco_have }
		#debug_log = eco_have_$TYPE$
		change_variable = { name = eco_mark add = 1 }
	}

	# build +1 in the best (highest-gdp) constructible state (additive: current + 1)
	ordered_scope_state = {
		order_by = gdp
		limit = { can_construct_building = $B$ }
		position = 0
		check_range_bounds = no
		set_variable = eco_$TYPE$_here
		set_variable = { name = eco_cur value = 0 }
		if = { limit = { has_building = $B$ } set_variable = { name = eco_cur value = b:$B$.level } }
		change_variable = { name = eco_cur add = 1 }
		create_building = { building = $B$ level = var:eco_cur reserves = 1 }
		#debug_log = eco_hit_$TYPE$
	}
}

# ---- cleanup: strip every eco state flag + the country counters --------------
eco_cleanup_flags = {
	remove_variable = eco_placed
	remove_variable = eco_S
	remove_variable = eco_bestlevel
	remove_variable = eco_mark
	remove_variable = eco_have
	remove_variable = eco_need
	remove_variable = eco_share
	every_scope_state = {
		limit = {
			OR = {
				has_variable = eco_champ_here
				has_variable = eco_support_here
				has_variable = eco_flavour_here
				has_variable = eco_best_here
				has_variable = eco_cur
			}
		}
		remove_variable = eco_champ_here
		remove_variable = eco_support_here
		remove_variable = eco_flavour_here
		remove_variable = eco_best_here
		remove_variable = eco_cur
	}
}
