# =============================================================================
# Top40EcoBoostMod — fire the Punjab eco-boost setup on game start.
# Additive on_actions sub-list pattern (does NOT add a 2nd `effect` to vanilla).
# on_actions do NOT hotload — changes need a return to main menu.
# Country-tag targeting is fine; state/geography selection inside the event is dynamic.
# =============================================================================

on_game_started_after_lobby = {
	on_actions = { top40_ecoboost_startup }
}

top40_ecoboost_startup = {
	effect = {
		c:PAN ?= { trigger_event = { id = punjab_ecoboost.1 } }
	}
}
