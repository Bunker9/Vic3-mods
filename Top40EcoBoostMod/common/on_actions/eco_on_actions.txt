# =============================================================================
# Top40EcoBoostMod — fire the eco-boost on game start.
# Additive on_actions sub-list pattern (merges with vanilla + other mods).
# on_actions do NOT hotload — return to main menu to apply changes.
# =============================================================================

on_game_started_after_lobby = {
	on_actions = { eco_startup }
}

eco_startup = {
	effect = {
		# 0) flag every target (var:eco_target + var:eco_tag_<TAG>) — runtime-safe dispatch so the
		#    yearly pulse never compares this = c:DEADTAG ("Invalid right side during comparison 'c'").
		eco_set_target_flags = yes
		# 1) grant the military-readiness techs to every target nation (the "for loop")
		every_country = {
			limit = { eco_is_target_country = yes }
			eco_grant_military_techs = yes
		}
		# 2) run each target nation's build pass via eco.2 (event scope sets ROOT = the country,
		#    which eco_generic_flavour + the per-nation setups rely on). Calling those effects
		#    straight from the on_action leaves ROOT unset → "root unset / invalid var" errors.
		every_country = {
			limit = { eco_is_target_country = yes }
			trigger_event = { id = eco.2 }
		}
	}
}

# Yearly Jan-01: re-run the SAME all-40 champ/support spread (eco.5 -> eco_pulse_<TAG>), so the
# spread reaches more states each year and newly-ANNEXED states get covered. Global pulse +
# every-target loop (mirrors eco_startup) — NOT per-country.
on_yearly_pulse = {
	on_actions = { eco_yearly }
}

eco_yearly = {
	effect = {
		every_country = {
			limit = { eco_is_target_country = yes }
			trigger_event = { id = eco.5 }
		}
	}
}
