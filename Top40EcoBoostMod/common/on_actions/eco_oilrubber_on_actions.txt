# =============================================================================
# Top40EcoBoostMod - oil/rubber reward hook (REDESIGN 2026-06-15).
# on_acquired_technology MERGES across mods (additive on_actions sub-list).
#   Root            = the country that just acquired a technology
#   scope:technology = the technology type acquired
# The first time the world unlocks pumpjacks (oil) / rubber_mastication (rubber), gift the
# tech + a seeded RGO to the themed beneficiaries. Global var guards make it ONCE per resource.
# =============================================================================

on_acquired_technology = {
	on_actions = { eco_oilrubber_on_acquired_tech }
}

eco_oilrubber_on_acquired_tech = {
	effect = {
		# OIL: first pumpjacks anywhere -> gift to Oman
		if = {
			limit = {
				has_technology_researched = pumpjacks      # Root just got it (or already had it)
				NOT = { has_global_variable = ecoor_oil_done }
			}
			set_global_variable = ecoor_oil_done
			c:OMA ?= { trigger_event = { id = ecoor.1 } }
		}

		# RUBBER: first rubber_mastication anywhere -> gift to Brazil + Siam (SIA = Thailand/Siam)
		if = {
			limit = {
				has_technology_researched = rubber_mastication
				NOT = { has_global_variable = ecoor_rubber_done }
			}
			set_global_variable = ecoor_rubber_done
			c:BRZ ?= { trigger_event = { id = ecoor.2 } }
			c:SIA ?= { trigger_event = { id = ecoor.2 } }
		}
	}
}
