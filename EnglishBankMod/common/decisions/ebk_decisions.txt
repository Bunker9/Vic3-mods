# EnglishBankMod decisions. Three, and no more.
# D1 and D2 are the same rescue seen from either side; D3 spends a fat pool at home.
# is_shown is deliberately LAX (cooldown only) so the entry stays visible and its
# requirements read out in the panel. Tighten only after an in-game pass.
# Every when_taken is custom_tooltip + cooldown + trigger_event, payload in the event.

# ---- D1: the debtor asks whoever can afford it -----------------------------

ebk_request_rescue = {

	is_shown = {
		NOT = { has_variable = ebk_rescue_cooldown_var }
	}

	possible = {
		ebk_trigger_in_distress_tt = yes
		ebk_trigger_has_rescuer_tt = yes
	}

	when_taken = {
		custom_tooltip = { text = ebk_request_rescue_tt }
		# Set here, not in the event: the gate must close on the click itself.
		set_variable = { name = ebk_rescue_cooldown_var value = yes days = 730 }
		trigger_event = { id = ebk_event.1 }
	}

	ai_chance = { value = 100 }
}

# ---- D2: the patron acts without being asked -------------------------------
# This is the arm that answers "overlord is flush, subject is broke".

ebk_underwrite_debtor = {

	is_shown = {
		NOT = { has_variable = ebk_underwrite_cooldown_var }
	}

	possible = {
		ebk_trigger_has_debtor_to_save_tt = yes
	}

	when_taken = {
		custom_tooltip = { text = ebk_underwrite_debtor_tt }
		set_variable = { name = ebk_underwrite_cooldown_var value = yes days = 730 }
		trigger_event = { id = ebk_event.2 }
	}

	ai_chance = { value = 100 }
}

# ---- D3: the pool pays for ten years of wages in one state -----------------
# The pool covering the price IS the gate, so there is no separate threshold.
# Taken yearly, richest uncovered state first, so a big country works down its
# states and then stacks the top ones while the ten year modifiers overlap.

ebk_national_uplift = {

	is_shown = {
		NOT = { has_variable = ebk_uplift_cooldown_var }
	}

	possible = {
		ebk_trigger_has_uplift_state_tt = yes
	}

	when_taken = {
		custom_tooltip = { text = ebk_national_uplift_tt }
		# One a year. The modifier lasts ten, so coverage accumulates.
		set_variable = { name = ebk_uplift_cooldown_var value = yes days = 365 }
		trigger_event = { id = ebk_event.4 }
	}

	ai_chance = { value = 100 }
}
